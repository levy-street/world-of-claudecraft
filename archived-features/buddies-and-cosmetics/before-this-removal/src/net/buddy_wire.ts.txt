// The client-side mirror of the buddy collection (IWorldBuddies): the four
// delta-guarded self-snapshot keys the server emits (server/buddy_wire.ts
// emitBuddySelfKeys), decoded through the catalog so a key or look a content
// change removed can never reach the HUD. A key omitted from a snapshot keeps
// the prior mirror, exactly like mntOwn. Pure: no DOM, no world state.

import { type BuddyKey, normalizeBuddyKey } from '../sim/content/buddies';
import { buddyCosmeticDef } from '../sim/content/buddy_cosmetics';

export interface BuddySelfMirror {
  owned: BuddyKey[];
  cosmetics: string[];
  equipped: Record<string, string>;
  pending: BuddyKey[];
}

export function emptyBuddySelfMirror(): BuddySelfMirror {
  return { owned: [], cosmetics: [], equipped: {}, pending: [] };
}

function keyList(value: unknown): BuddyKey[] | null {
  if (!Array.isArray(value)) return null;
  return (value as unknown[])
    .map((k) => normalizeBuddyKey(typeof k === 'string' ? k : ''))
    .filter((k): k is BuddyKey => k !== '');
}

/** Fold one self snapshot into the mirror. Returns the same object when the
 *  snapshot carried none of the four keys, so a caller can keep identity. */
export function decodeBuddySelf(
  s: Record<string, unknown>,
  prev: BuddySelfMirror,
): BuddySelfMirror {
  const owned = keyList(s.budOwn);
  const pending = keyList(s.budPend);
  const cosmetics = Array.isArray(s.budCos)
    ? (s.budCos as unknown[]).filter(
        (id): id is string => typeof id === 'string' && buddyCosmeticDef(id) !== null,
      )
    : null;
  let equipped: Record<string, string> | null = null;
  if (s.budEq !== undefined && s.budEq !== null && typeof s.budEq === 'object') {
    equipped = {};
    for (const [key, id] of Object.entries(s.budEq as Record<string, unknown>)) {
      const norm = normalizeBuddyKey(key);
      const cosmetic = typeof id === 'string' ? buddyCosmeticDef(id) : null;
      if (norm && cosmetic?.buddy === norm) equipped[norm] = cosmetic.id;
    }
  }
  if (!owned && !pending && !cosmetics && !equipped) return prev;
  return {
    owned: owned ?? prev.owned,
    cosmetics: cosmetics ?? prev.cosmetics,
    equipped: equipped ?? prev.equipped,
    pending: pending ?? prev.pending,
  };
}
